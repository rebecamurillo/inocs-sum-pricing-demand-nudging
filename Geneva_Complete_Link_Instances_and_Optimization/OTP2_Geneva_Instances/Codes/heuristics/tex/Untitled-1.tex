\section{Algorithm Description}

The proposed matheuristic consists of five distinct phases: (1) competitive initial selection, (2) candidate heap construction, (3) restricted master problem initialization, (4) forward greedy addition, and (5) backward elimination. Each phase is designed to efficiently explore the solution space while respecting the bilevel structure.

\begin{algorithm}[htbp]
\caption{Greedy Fix-and-Optimize Matheuristic for Bilevel Bike-Sharing Network Design}
\label{alg:greedy_fix_optimize}
\begin{algorithmic}[1]
\Require Parsed network data, demand matrix $Q$, parameters $L$, $K_{PT}$, $mult_{A1}$
\Ensure Optimal arc selection and pricing solution

\State \textbf{Phase I: Initial Selection Construction}
\State $\mathcal{S} \leftarrow$ \Call{InitialSelection}{$L$, $K_{PT}$, $mult_{A1}$}
\Comment{Enhanced competitive selection}
\For{each OD pair $(o,d)$}
    \State Compute best car utility $u^{car}_{od}$ and best PT utility $u^{PT}_{od}$
    \State $\tau_{od} \leftarrow \min(u^{car}_{od}, u^{PT}_{od})$ \Comment{Competitive threshold}
    \State Score PT+bike arcs at lower bound $\bar{P}$: $u^{A1}_{oda} = \mathcal{U}_{A1}(o,d,a,\bar{P})$
    \State $\mathcal{C}_{od} \leftarrow \{a : u^{A1}_{oda} < \tau_{od}\}$ \Comment{Competitive PT+bike arcs}
    \If{$|\mathcal{C}_{od}| > 0$}
        \State Select $\min(\max(L, mult_{A1} \cdot L), |\mathcal{C}_{od}|)$ best PT+bike arcs
        \State Select $\max(1, K_{PT} - 1)$ best PT-only arcs \Comment{Prioritize PT+bike}
    \Else
        \State Standard selection: $\max(L, mult_{A1} \cdot L)$ PT+bike, $K_{PT}$ PT-only
    \EndIf
\EndFor

\State \textbf{Phase II: Candidate Heap Construction}
\State $\mathcal{H} \leftarrow \emptyset$ \Comment{Global priority queue}
\For{each OD pair $(o,d)$ in parallel}
    \State Compute benchmark $\beta_{od} \leftarrow \min(u^{car}_{od}, u^{PT}_{od})$
    \For{each PT-only arc $a \notin \mathcal{S}_{od}^{A2}$}
        \State $s \leftarrow \mathcal{U}_{A2}(o,d,a)$ \Comment{Standard PT disutility}
        \State \Call{HeapPush}{$\mathcal{H}$, $(s, (o,d,a), \text{``A2''})$}
    \EndFor
    \For{each PT+bike arc $a \notin \mathcal{S}_{od}^{A1}$}
        \State $s \leftarrow \mathcal{U}_{A1}(o,d,a,\bar{P}) - \beta_{od}$ \Comment{Gap-based scoring}
        \State \Call{HeapPush}{$\mathcal{H}$, $(s, (o,d,a), \text{``A1''})$}
    \EndFor
\EndFor

\State \textbf{Phase III: Restricted Master Problem Initialization}
\State \Call{BuildLP}{$\mathcal{S}$} \Comment{Construct bilevel LP with initial selection}
\State $\theta^* \leftarrow$ \Call{Optimize}{} \Comment{Solve base problem}
\State $trials \leftarrow 0$, $patience \leftarrow 0$, $added \leftarrow 0$

\State \textbf{Phase IV: Forward Greedy Addition}
\While{$\mathcal{H} \neq \emptyset$ \textbf{and} $added < max_{add}$ \textbf{and} $trials < max_{trials}$ \textbf{and} $patience < max_{patience}$}
    \State $(score, (o,d,a), type) \leftarrow$ \Call{HeapPop}{$\mathcal{H}$}
    \State $\theta_{trial} \leftarrow$ \Call{TrialAddValue}{$o,d,a,type$} \Comment{Model copy evaluation}
    \State $trials \leftarrow trials + 1$
    \If{$\theta_{trial}$ is \textbf{None}}
        \State $patience \leftarrow patience + 1$
        \State \textbf{continue}
    \EndIf
    \State $\Delta \leftarrow \frac{\theta_{trial} - \theta^*}{|\theta^*|}$ \Comment{Relative improvement}
    \If{$\Delta > \epsilon_{add}$} \Comment{Accept if sufficient improvement}
        \If{$type = \text{``A2''}$}
            \State \Call{AddPTOnly}{$o,d,a$}
        \Else
            \State \Call{AddPTBike}{$o,d,a$}
        \EndIf
        \State $\theta^* \leftarrow$ \Call{Optimize}{}
        \State $added \leftarrow added + 1$, $patience \leftarrow 0$
    \Else
        \State $patience \leftarrow patience + 1$
    \EndIf
\EndWhile

\State \textbf{Phase V: Backward Elimination}
\State $\mathcal{R} \leftarrow$ \Call{RankA1ByContribution}{} \Comment{Sort by flow and contribution}
\State $drops \leftarrow 0$, $trials \leftarrow 0$, $patience \leftarrow 0$
\For{each $(q_{val}, contrib, (o,d,a)) \in \mathcal{R}$}
    \If{$drops \geq max_{drops}$ \textbf{or} $trials \geq max_{trials}$ \textbf{or} $patience \geq max_{patience}$}
        \State \textbf{break}
    \EndIf
    \State $\theta_{trial} \leftarrow$ \Call{TrialDropValue}{$o,d,a$} \Comment{Evaluate removal}
    \State $trials \leftarrow trials + 1$
    \If{$\theta_{trial}$ is \textbf{None}}
        \State $patience \leftarrow patience + 1$
        \State \textbf{continue}
    \EndIf
    \State $\Delta \leftarrow \theta_{trial} - \theta^*$
    \If{$\Delta \geq |\theta^*| \cdot \epsilon_{drop}$ \textbf{or} $(q_{val} \leq \epsilon_q \textbf{ and } \Delta > -|\theta^*| \cdot \epsilon_{drop})$}
        \State \Call{DeactivatePTBike}{$o,d,a$} \Comment{Remove weak arc}
        \State $\theta^* \leftarrow$ \Call{Optimize}{}
        \State $drops \leftarrow drops + 1$, $patience \leftarrow 0$
    \Else
        \State $patience \leftarrow patience + 1$
    \EndIf
\EndFor

\State \textbf{return} $\theta^*$, solution vectors
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[htbp]
\caption{Supporting Procedures}
\label{alg:supporting_procedures}
\begin{algorithmic}[1]

\Procedure{TrialAddValue}{$o, d, a, type$}
    \State $\mathcal{M}_{copy} \leftarrow$ \Call{CopyModel}{}
    \State Add variables and constraints for arc $(o,d,a)$ of given $type$
    \If{$type = \text{``A2''}$}
        \State Add: $q_{oda} \in [0, Q_{od}]$, dual feasibility, demand, strong duality constraints
    \Else \Comment{PT+bike arc}
        \State Add: $p_{oda} \in [\bar{P}, \bar{P}^+]$, $q_{oda} \in [0, Q_{od}]$, $z_{oda}$ (McCormick)
        \State Add: capacity, dual feasibility, demand, strong duality constraints
    \EndIf
    \State \textbf{return} \Call{Optimize}{$\mathcal{M}_{copy}$}
\EndProcedure

\Procedure{TrialDropValue}{$o, d, a$}
    \State $\mathcal{M}_{copy} \leftarrow$ \Call{CopyModel}{}
    \State Fix $q_{oda} = p_{oda} = z_{oda} = 0$ \Comment{Deactivate arc}
    \State Relax dual constraint: $\gamma_{od} \leq M$ (large $M$)
    \State \textbf{return} \Call{Optimize}{$\mathcal{M}_{copy}$}
\EndProcedure

\end{algorithmic}
\end{algorithm}

% ================== MAIN GREEDY PIPELINE (BREAKABLE) ==================
\begin{breakablealgorithm}[Greedy Fix--and--Optimize Matheuristic]
\label{alg:greedy_fix_optimize}
\begin{algorithmic}[1]
\Require Parsed instance $(\mathcal{W}, \mathcal{A}^{PT}, \mathcal{A}^{A1})$, demands $Q$, parameters $L$, $K_{\!PT}$, $mult_{A1}$, pricing bounds $[\bar P, \bar P^{+}]$
\Ensure Selected alternatives and prices; objective value $\theta^\star$

\State \textbf{Phase I (Competitive Initial Selection)}
\For{each $(o,d)\in\mathcal{W}$}
  \State Compute best car utility $u^{car}_{od}$ and best PT utility $u^{PT}_{od}$
  \State $\tau_{od}\gets \min(u^{car}_{od},u^{PT}_{od})$
  \State Score PT+bike at $\bar P$: $u^{A1}_{oda}$
  \State $\mathcal{C}_{od}\gets\{a\in\mathcal{A}^{A1}_{od}: u^{A1}_{oda}<\tau_{od}\}$
  \If{$|\mathcal{C}_{od}|>0$} \Comment{PT+bike competitive}
    \State Select $\min\{\max(L,\,mult_{A1}\!\cdot\!L),\,|\mathcal{C}_{od}|\}$ best $A1$
    \State Select $\max(1, K_{\!PT}-1)$ best PT-only
  \Else
    \State Select $\max(L,\,mult_{A1}\!\cdot\!L)$ best $A1$ and $K_{\!PT}$ best PT-only
  \EndIf
\EndFor
\State Let $\mathcal{S}$ be the union of selected $A1$/$A2$ per OD

\Statex
\State \textbf{Phase II (Global Candidate Heap)}
\For{each $(o,d)\in\mathcal{W}$ \textbf{in parallel}}
  \State $\beta_{od}\gets \min(u^{car}_{od},u^{PT}_{od})$
  \For{each $a\in\mathcal{A}^{PT}_{od}\setminus\mathcal{S}$}
    \State $s\gets\mathcal{U}_{A2}(o,d,a)$; push $(s,(o,d,a),A2)$ to heap
  \EndFor
  \For{each $a\in\mathcal{A}^{A1}_{od}\setminus\mathcal{S}$}
    \State $s\gets \mathcal{U}_{A1}(o,d,a,\bar P)-\beta_{od}$; push $(s,(o,d,a),A1)$ to heap
  \EndFor
\EndFor

\Statex
\State \textbf{Phase III (Restricted Master Initialization)}
\State Build the incremental LP with all items in $\mathcal{S}$
\State Solve to obtain $\theta^\star$ (free method), then set reopt method to Dual Simplex

\Statex
\State \textbf{Phase IV (Forward Greedy Addition)}
\State $added\gets 0$, $trials\gets 0$, $patience\gets 0$
\While{heap not empty \textbf{and} $added<max_{\!add}$ \textbf{and} $trials<max_{\!trials}$ \textbf{and} $patience<max_{\!patience}$}
  \State Pop $(score,(o,d,a),type)$ from heap
  \State $\theta^{trial}\gets$ \Call{TrialAddValue}{$o,d,a,type$}
  \State $trials\gets trials+1$
  \If{$\theta^{trial}$ is undefined}
     \State $patience\gets patience+1$; \textbf{continue}
  \EndIf
  \State $\Delta \gets \dfrac{\theta^{trial}-\theta^\star}{|\theta^\star|}$
  \If{$\Delta>\varepsilon_{\!add}$}
     \State Add $(o,d,a)$ of $type$ to the live LP; reoptimize to update $\theta^\star$
     \State $added\gets added+1$; $patience\gets 0$
  \Else
     \State $patience\gets patience+1$
  \EndIf
\EndWhile

\Statex
\State \textbf{Phase V (Backward Elimination)}
\State Rank live $A1$ by increasing $(q\text{-flow},\,\text{contribution})$
\State $drops\gets 0$, $trials\gets 0$, $patience\gets 0$
\For{each $(o,d,a)$ in ranked order}
  \If{$drops\geq max_{\!drops}$ \textbf{or} $trials\geq max_{\!trials}$ \textbf{or} $patience\geq max_{\!patience}$}
    \State \textbf{break}
  \EndIf
  \State $\theta^{trial}\gets$ \Call{TrialDropValue}{$o,d,a$}
  \State $trials\gets trials+1$
  \If{$\theta^{trial}$ is undefined}
     \State $patience\gets patience+1$; \textbf{continue}
  \EndIf
  \State $\Delta\gets \theta^{trial}-\theta^\star$
  \If{$\Delta \ge |\theta^\star|\varepsilon_{\!drop}$ \textbf{or} $(q_{oda}\le \varepsilon_q \ \textbf{and}\  \Delta > -|\theta^\star|\varepsilon_{\!drop})$}
     \State Deactivate $(o,d,a)$ in the live LP; reoptimize to update $\theta^\star$
     \State $drops\gets drops+1$; $patience\gets 0$
  \Else
     \State $patience\gets 1+patience$
  \EndIf
\EndFor

\State \Return $\theta^\star$ and the final solution
\end{algorithmic}
\end{breakablealgorithm}


% ================== SUPPORTING PROCEDURES (BREAKABLE) ==================
\begin{breakablealgorithm}[Supporting Procedures]
\begin{algorithmic}[1]
\Procedure{TrialAddValue}{$o,d,a,type$}
  \State Create a copy of the current LP
  \If{$type$ is PT-only}
     \State Add $q_{oda}\in[0,Q_{od}]$; add dual-feasibility, demand, strong-duality rows
     \State Set $q$-variable objective coefficient as in the master model
  \Else \Comment{PT+bike}
     \State Add $p_{oda}\in[\bar P,\bar P^+]$, $q_{oda}\in[0,Q_{od}]$, and $z_{oda}$ with McCormick envelopes
     \State Add capacity links, dual-feasibility, demand, strong-duality rows
     \State Set objective coefficients for $q$ and $z$ as in the master model
  \EndIf
  \State \Return optimal objective of the copied model (Dual Simplex)
\EndProcedure

\Procedure{TrialDropValue}{$o,d,a$}
  \State Create a copy of the current LP
  \State Fix $q_{oda}=0$, and if present $p_{oda}=z_{oda}=0$; relax the corresponding dual row with big-$M$
  \State \Return optimal objective of the copied model (Dual Simplex)
\EndProcedure
\end{algorithmic}
\end{breakablealgorithm}






\section{Computational Complexity and Performance}

The algorithm's complexity is dominated by the number of bilevel LP solves performed during the forward and backward phases. Each trial evaluation requires solving a copy of the master problem, leading to a complexity of $O(T \cdot C_{LP})$ where $T$ is the total number of trials and $C_{LP}$ is the complexity of solving the bilevel LP.

The use of model copying for trial evaluations provides numerical stability and avoids basis-related issues that can occur with direct model modifications. The patience mechanism prevents excessive exploration in unpromising regions of the solution space.
