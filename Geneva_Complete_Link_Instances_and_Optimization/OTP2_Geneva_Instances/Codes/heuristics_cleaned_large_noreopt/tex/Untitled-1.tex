% ---------- Notation ----------
\paragraph{Notation (common to all algorithms).}
Let $\mathcal{W}$ denote the set of OD pairs, with demand $Q_{od}$ for $(o,d)\in\mathcal{W}$.
For each $(o,d)$, let $\mathcal{C}_{od}$ be the set of car alternatives,
$\mathcal{A}^{(2)}_{od}$ PT-only alternatives, and $\mathcal{A}^{(1)}_{od}$ PT+Bike alternatives.
Let $\mathcal{I}$ be the set of bike stations, and for each PT+Bike arc $a\in\mathcal{A}^{(1)}_{od}$,
$\delta_{ia}\in\{-1,0,+1\}$ is the arc--station incidence (-1 for departure, +1 for arrival).
Deterministic disutilities at bike price $p$ are:
\[
V^{(1)}_{od}(a;p)\quad\text{(PT+Bike)},\qquad
V^{(2)}_{od}(a)\quad\text{(PT-only)},\qquad
V^{(c)}_{od}(b)\quad\text{(car)}.
\]
From the current restricted LP solution, we read:
$\gamma_{od}$ (dual on OD balance),
$\lambda_i$ (effective station shadow price: upper minus lower),
$U_i\ge 0$ (residual \emph{arrival} capacity / dock slack),
$L_i\ge 0$ (residual \emph{departure} capacity / inventory slack).
Per-OD caps: $L$ (PT+Bike), $K_{\mathrm{PT}}$ (PT-only), multiplier $\mathrm{mult}_{A1}\ge 0$.
Global batch size $M$.
Station dual weight $\mu_{\mathrm{st}}\in[0,1]$.

Define the \emph{dual-aware gap} for a PT+Bike arc $a\in\mathcal{A}^{(1)}_{od}$ at $p=P_{\mathrm{bar}}$:
\[
\mathrm{gap}_{od}(a)\;=\;\Big(V^{(1)}_{od}(a;P_{\mathrm{bar}})
\;+\;\mu_{\mathrm{st}}\sum_{i\in\mathcal{I}}\lambda_i\,\delta_{ia}\Big) \;-\; \gamma_{od},
\qquad
w_{od}(a)\;:=\; -\,\mathrm{gap}_{od}(a),
\]
where $w_{od}(a)$ is a value proxy (larger is better).
A conservative per-arc flow proxy is
\[
\widehat q_{od}(a)\;=\;
\max\Bigl(0,\,\min\bigl\{Q_{od},\ \min_{\delta_{ia}>0} U_i,\ \min_{\delta_{ia}<0} L_i\bigr\}\Bigr).
\]

\paragraph{Notation.}
\begin{description}
\item[$\mathcal W$] Set of OD pairs; elements $(o,d)$.
\item[$\mathcal A_1^{(o,d)}$] PT+Bike (A1) alternatives for $(o,d)$.
\item[$\mathcal A_2^{(o,d)}$] PT-only (A2) alternatives for $(o,d)$.
\item[$I$] Set of bike stations; indices $i$.
\item[$Q_{od}$] Demand of OD pair $(o,d)$.
\item[$C_i$] Capacity (docks) at station $i$; $s_i$ is current inventory (bikes).
\item[$\delta_{ia}$] Station-incidence of A1 arc $a$: $+1$ if $a$ arrives at $i$, $-1$ if $a$ departs, $0$ otherwise.
\item[$k_a$] Number of bike \emph{legs} in A1 arc $a$ (0, 1 or 2). % new
\item[$p_a$] Bike price decision (continuous) on A1 arc $a$, bounded in $[P_{\min},P_{\max}]$.
\item[$q_a^{(o,d)}$] Chosen flow on arc $a\in\mathcal A_1^{(o,d)} \cup \mathcal A_2^{(o,d)} \cup \mathcal A_{\mathrm{car}}^{(o,d)}$.
\item[$z_a$] McCormick auxiliary representing $p_a \cdot q_a^{(o,d)}$ for $a\in\mathcal A_1^{(o,d)}$.
\item[$V_a^{(o,d)}(p)$] System disutility of alternative $a$ for $(o,d)$; for A1 it includes $k_a \beta^{\mathrm{bike}}_{\mathrm{price}}\,p$.
\item[$\gamma_{od}$] Dual variable of the OD demand-conservation row for $(o,d)$ in the single-level LP (the “choice” potential).
\item[$\pi^{\mathrm{up}}_i\!\ge 0$] Dual of the station \emph{upper} row $\sum\delta_{ia} q_a^{(o,d)} \le C_i - s_i$.
\item[$\pi^{\mathrm{lo}}_i\!\le 0$] Dual of the station \emph{lower} row $\sum\delta_{ia} q_a^{(o,d)} + s_i \ge 0$.
\item[$\lambda_i$] Effective station shadow price $\lambda_i := \pi^{\mathrm{up}}_i + \pi^{\mathrm{lo}}_i$.
\item[$U_i, L_i$] Current slacks of the upper and lower station rows, respectively.
\item[$\overline V_a^{(o,d)}$] Disutility evaluated at a ranking proxy price $\bar P$ (for A1: includes $k_a \beta^{\mathrm{bike}}_{\mathrm{price}}\bar P$).
\item[$\textsf{score}(a)$] Surrogate reduced-cost score used in D1:
$\textsf{score}(a)=(\overline V_a^{(o,d)}-\gamma_{od}) + \mu_{\mathrm{st}}\sum_i \lambda_i \delta_{ia}$.
\item[$\widehat q_a$] Conservative flow proxy for arc $a$:
$\widehat q_a = \min\bigl(Q_{od},\,\min_{\delta_{ia}>0} U_i,\,\min_{\delta_{ia}<0} L_i\bigr)$.
\end{description}


% ---------- Algorithm 1 ----------
\paragraph{Algorithm 1 — Dual-Aware Batched Forward Selection (DABFS).}
\emph{Inputs:} $\mathcal{W}$, arcs, utilities, $P_{\mathrm{bar}},P_{\mathrm{up}}$, $L,K_{\mathrm{PT}},\mathrm{mult}_{A1}$, $M,\mu_{\mathrm{st}}$.
\emph{Output:} accepted PT+Bike/PT-only arcs and the incumbent LP solution.
\begin{enumerate}
\item \textbf{Initial selection (per OD).}
  \begin{enumerate}
  \item Compute $V^{(c),\star}_{od}:=\min_{b\in\mathcal{C}_{od}} V^{(c)}_{od}(b)$.
  \item Score all PT+Bike arcs at $P_{\mathrm{bar}}$ and split
  \[
    \mathrm{Comp}_{od}:=\{a: V^{(1)}_{od}(a;P_{\mathrm{bar}}) < V^{(c),\star}_{od}\},\quad
    \mathrm{NonComp}_{od}:=\mathcal{A}^{(1)}_{od}\setminus\mathrm{Comp}_{od}.
  \]
  \item Base cap $C^{(1)}_{od}:=\lceil \mathrm{mult}_{A1}\cdot L\rceil$. If $|\mathrm{Comp}_{od}|>C^{(1)}_{od}$,
  surge to $C^{(1)}_{od}\leftarrow \min\{L,\,|\mathrm{Comp}_{od}|\}$.
  Select the $C^{(1)}_{od}$ best PT+Bike by $V^{(1)}$.
  \item Select $C^{(2)}_{od}:=\min\{K_{\mathrm{PT}},\,|\mathcal{A}^{(2)}_{od}|\}$ best PT-only by $V^{(2)}$.
  \end{enumerate}
\item \textbf{Build and solve the restricted LP.} Solve to optimality; store $\gamma,\lambda,U,L$.
\item \textbf{Main forward loop} (up to max batches):
  \begin{enumerate}
  \item \textbf{Candidate scoring.} For all not-present PT+Bike arcs, compute $\mathrm{gap}_{od}(a),w_{od}(a),\widehat q_{od}(a)$.
  \item \textbf{Batching.} Use one of D1–D5 to produce a batch $\mathcal{B}\subseteq\{(o,d,a)\}$ with $|\mathcal{B}|\le M$.
  \item \textbf{Per-OD cap enforcement.} Greedily discard items so that for each $(o,d)$,
  \[
  \#\mathrm{present}^{(1)}_{od} + \#\bigl(\mathcal{B}\cap\{od\}\bigr) \le L.
  \]
  If $\mathcal{B}=\emptyset$, terminate.
  \item \textbf{Trial solve with $\varepsilon$ tie-break.}
  On a model copy, add $\mathcal{B}$; set objective coefficients as in the live model and add a tiny $\varepsilon>0$
  \emph{only} to the new $q$-coefficients. Solve; record $z^{\mathrm{trial}}$ and 
  $q^{\mathrm{new}}:=\sum_{(o,d,a)\in\mathcal{B}} q_{od}(a)$.
  \item \textbf{Acceptance (neutral-or-better).} With incumbent $z^{\mathrm{base}}$, tolerance
  $\tau=\max\{\varepsilon_{\mathrm{abs}},\,\varepsilon_{\mathrm{rel}}\cdot\max(1,|z^{\mathrm{base}}|)\}$,
  accept if
  \[
  z^{\mathrm{trial}} \ge z^{\mathrm{base}}+\tau
  \quad\text{or}\quad
  \bigl(|z^{\mathrm{trial}}-z^{\mathrm{base}}|\le \tau\ \wedge\ q^{\mathrm{new}}>0\bigr).
  \]
  \item \textbf{Apply \& reoptimize (if accepted).} Add $\mathcal{B}$ to the live LP and re-solve; update $\gamma,\lambda,U,L$.
  If rejected, terminate.
  \end{enumerate}
\end{enumerate}

\noindent\emph{Properties.} The live objective is non-decreasing across accepted batches;
feasibility is preserved; selection is dual- and station-budget-aware.

% ---------- D1 ----------
\paragraph{D1 — Token-Budget Greedy with Complementary Pairing.}
\emph{Inputs:} candidates $\mathcal{E}$ with $(w_e,\widehat q_e,\delta_{ie})$, budgets $U,L$, lookahead $H$, safety $s\ge 1$, batch size $M$.
\emph{Output:} $\mathcal{B}$ with $|\mathcal{B}|\le M$.
\begin{enumerate}
\item Sort $\mathcal{E}$ by $v_e:=w_e\cdot \max(\widehat q_e,1)$ (desc).
Partition $\mathcal{E}=\mathcal{E}^+\cup\mathcal{E}^0$ with $\widehat q_e>0$ vs.\ $=0$.
\item Initialize residual budgets $U'_i\gets U_i$, $L'_i\gets L_i$, and $\mathcal{B}\gets\emptyset$.
\item Scan $\mathcal{E}^+$ in order. For each $e$:
\begin{itemize}
\item \emph{Solo fit:} if $s\,\widehat q_e\,\delta_{ie}\in[-L'_i,U'_i]$ for all $i$, add $e$ to $\mathcal{B}$ and update
$U'_i\leftarrow U'_i-\max(0,s\,\widehat q_e\,\delta_{ie})$, $L'_i\leftarrow L'_i-\max(0,-s\,\widehat q_e\,\delta_{ie})$.
\item \emph{Else, pairing:} search among the next $H$ items (from $\mathcal{E}^+$ and $\mathcal{E}^0$) for $f\ne e$
such that $s(\widehat q_e\delta_{ie}+\widehat q_f\delta_{if})\in[-L'_i,U'_i]$ for all $i$.
If found, add both (respecting $M$) and update budgets with the \emph{joint} load.
\end{itemize}
\item If tokens remain, greedily fill with the best fitting items.
\item Return $\mathcal{B}$ (truncate to $M$ if needed).
\end{enumerate}

% ---------- D2 ----------
\paragraph{D2 — Two-Sided Budgeted Knapsack (LP).}
\emph{Inputs:} $\mathcal{E}$ with $(w_e,\widehat q_e,\delta_{ie})$, budgets $U,L$, softness $\alpha\in(0,1]$, batch size $M$.
\emph{Output:} $\mathcal{B}$ with $|\mathcal{B}|\le M$.
\[
\begin{aligned}
\max_{0\le x_e\le 1}\quad & \sum_{e\in\mathcal{E}} w_e\,x_e \\
\text{s.t.}\quad
& -\alpha\,L_i \;\le\; \sum_{e\in\mathcal{E}} \delta_{ie}\,\widetilde q_e\,x_e \;\le\; \alpha\,U_i
\quad \forall i\in\mathcal{I},\\
& \sum_{e\in\mathcal{E}} x_e \;\le\; M,
\end{aligned}
\qquad\text{where}\quad
\widetilde q_e:=\max(\widehat q_e,1).
\]
Pick the $M$ largest $x_e$.

% ---------- D3 ----------
\paragraph{D3 — Reversing-Pair Cycle Greedy.}
\emph{Inputs:} edges $u\to v$ per arc with weight $w_e$, batch size $M$.
\emph{Output:} $\mathcal{B}$ with $|\mathcal{B}|\le M$.
\begin{enumerate}
\item Sort edges by $w_e$ (desc).
\item Greedily take reversing pairs: for unused $u\to v$, if an unused $v\to u$ exists, add both.
\item Fill any remaining slots with best unused edges; truncate to $M$.
\end{enumerate}

% ---------- D4 ----------
\paragraph{D4 — Exact Station-Balance Pool (LP).}
\emph{Inputs:} pool $\mathcal{E}$ (optionally top-$N$ by $w_e$), batch size $M$.
\emph{Output:} $\mathcal{B}$ with $|\mathcal{B}|\le M$.
\[
\begin{aligned}
\max_{0\le x_e\le 1}\quad & \sum_{e\in\mathcal{E}} w_e\,x_e \\
\text{s.t.}\quad
& \sum_{e\in\mathcal{E}} \delta_{ie}\,x_e \;=\;0 \quad \forall i\ \text{with some }\delta_{ie}\ne 0,\\
& \sum_{e\in\mathcal{E}} x_e \;\le\; M.
\end{aligned}
\]
Pick the $M$ largest $x_e$.

% ---------- D5 ----------
\paragraph{D5 — $k$-Cycle Hunt on the Station Graph.}
\emph{Inputs:} directed station graph induced by $\mathcal{E}$, DFS depth $D$, batch size $M$.
\emph{Output:} $\mathcal{B}$ with $|\mathcal{B}|\le M$.
\begin{enumerate}
\item Enumerate simple directed cycles up to length $D$ from promising starts.
\item Score cycle $C$ by $\sum_{e\in C} w_e$; sort by score.
\item Greedily pack disjoint arcs from top cycles into $\mathcal{B}$ until $|\mathcal{B}|=M$.
\end{enumerate}

% ---------- Trial ε tie-break & live objective ----------
\paragraph{Trial $\boldsymbol{\varepsilon}$ tie-break and live objective consistency.}
In the trial LP (model copy), add $\varepsilon>0$ \emph{only} to the objective coefficients of newly added $q$-variables,
so the perturbed objective is
\[
z^{\mathrm{trial}}_{\varepsilon}
\;=\;
z^{\mathrm{true}} \;+\; \varepsilon\sum_{(o,d,a)\in\mathcal{B}} q_{od}(a),
\]
which breaks degeneracy in favor of positive new flow.
For acceptance based on the \emph{true} objective, subtract the $\varepsilon$-term:
\[
z^{\mathrm{trial}}_{\mathrm{true}}
\;=\;
z^{\mathrm{trial}}_{\varepsilon}
\;-\;
\varepsilon\,\sum_{(o,d,a)\in\mathcal{B}} q_{od}(a),
\]
and apply the neutral-or-better rule with $q^{\mathrm{new}}=\sum_{(o,d,a)\in\mathcal{B}} q_{od}(a)$.
In the live model we do \emph{not} include $\varepsilon$; the objective may tie, but the accepted columns
expand the feasible set and can unlock subsequent improvements.
